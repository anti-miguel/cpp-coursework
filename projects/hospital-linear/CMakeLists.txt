if(BUILD_TESTING AND COURSE_TEST_FILE)
    if(NOT EXISTS "${COURSE_TEST_FILE}")
        message(FATAL_ERROR "COURSE_TEST_FILE does not exist")
    endif()
    find_package(GTest CONFIG REQUIRED)
    coursework_program(ProyectoAED1_test "${COURSE_TEST_FILE}")
    target_link_libraries(ProyectoAED1_test PRIVATE hospital_linear GTest::gtest_main)
    add_test(NAME hospital_course_tests COMMAND ProyectoAED1_test)
    set_tests_properties(hospital_course_tests PROPERTIES TIMEOUT 30)
endif()
